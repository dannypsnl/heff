module std.http.server (Request, header, plain, json, notAllowed, segments, queries, queryAll, query, parseHead, BodyIs(..), HeadRead(..), Got(..), Wanted(..), requestIn, Chunks, chunks, Dechunked(..), feedChunks, reason, responseBytes, chunk, Head, streamedHead, takeoverHead, keptAlive, After(..), Taken, Reply(..), serveConnection, serveUntil, serve)

import co
import exception
import resource
import bytes as Bytes
import custodian as Custodian
import json as JSON
import std.duration as Duration
import std.http as HTTP
import std.net as Net
import std.time as Time

import stream as Stream

type Request = { method : String, target : String, version : String, headers : List (String, String), body : Bytes, peer : Net.Address }
header : String -> Request -> Maybe String
plain : Int -> String -> HTTP.Response
json : Int -> JSON.Json -> HTTP.Response
notAllowed : List String -> HTTP.Response
parseHead : Net.Address -> String -> Maybe Request
segments : Request -> List String
queries : Request -> List (String, String)
queryAll : String -> Request -> List String
query : String -> Request -> Maybe String
data BodyIs = Length Int | Chunked | Refuse Int
type Chunks = { at : ChunkAt, line : Bytes, parts : List Bytes, taken : List Bytes, size : Int, trailer : Int }
chunks : Chunks
data Dechunked = BodyRead Bytes Bytes | ChunksToCome Chunks | ChunksRefused Int
feedChunks : Chunks -> Bytes -> Dechunked
data HeadRead = HeadAt Bytes Int | HeadTooLong | HeadEnded
data Got = Asked Request Bytes | Refused Int | Ended
data Wanted = MoreHead Int | MoreBody Request Int | MoreChunks Request Chunks
requestIn : {from : Int := 0} -> Net.Address -> Bytes -> Either Wanted Got
reason : Int -> String
responseBytes : String -> Bool -> HTTP.Response -> Bytes
chunk : Bytes -> Bytes
type Head = { status : Int, headers : List (String, String) }
streamedHead : String -> Bool -> Head -> Bytes
takeoverHead : Head -> Bytes
keptAlive : Request -> Bool
data After = Next Bytes | Done Bool
type Taken s = { request : Request, connection : With s Net.TcpStream, rest : Bytes }
data Reply (e : HFunctor) = Whole HTTP.Response | Streamed Head (Co (Stream.yielding Bytes <+> e) Unit) | TakenOver Head ({s : Type} -> Taken s -> Co e Unit)
serveConnection : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> (Request -> Bool) -> (Request -> Co e (Reply e)) -> With s Net.TcpStream -> Co e Bool
serveUntil : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> {{cu : Custodian.custodian <: e}} -> (Request -> Bool) -> With s Net.TcpListener -> (Request -> Co e (Reply e)) -> Co e Unit
serve : {e : HFunctor} -> {s : Type} -> {idle : Int := 30000} -> {headerWithin : Int := 30000} -> {{nt : Net.net <: e}} -> {{clock : Time.time <: e}} -> {{ex : exception Net.NetError <: e}} -> {{cu : Custodian.custodian <: e}} -> With s Net.TcpListener -> (Request -> Co e (Reply e)) -> Co e Unit
